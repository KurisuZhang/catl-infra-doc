

### 一、剪枝核心引擎：掩码（Mask）的生成与应用

剪枝的本质不是“删除”参数，而是**在权重矩阵上叠加一个二进制掩码（Mask）** ，掩码为0的位置表示该权重被“剪掉”（前向传播时强制为0）。这种方式便于恢复和微调。



```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from torch.utils.data import DataLoader
from torchvision import datasets, transforms
import copy
import numpy as np

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"使用设备: {device}")

# ===================== 1. 核心剪枝工具函数 =====================

def apply_mask(weight, mask):
    """应用掩码：将 mask=0 的位置权重强制置零"""
    return weight * mask

def compute_sparsity(weight, mask=None):
    """计算稀疏度（零元素占比）"""
    if mask is None:
        mask = (weight != 0)
    return 1.0 - (mask.sum().item() / mask.numel())
```

### 二、手算推演：Magnitude（幅值） vs Wanda 的终极对决

我们直接复现第5课补充材料中的那个**决定性例子**——输入中有个巨无霸 `100`，幅值剪枝会犯致命错误。



```python
print("="*60)
print("【实验1：Magnitude（幅值） vs Wanda（激活感知）的微观对决】")

# 构造第5课的经典场景：权重矩阵 (2行 x 3列)
W = torch.tensor([
    [0.5,  0.1,  10.0],   # 输出A
    [0.2,  9.8,   0.3]    # 输出B
])
# 输入激活：第1个特征巨大（100），其余很小
X = torch.tensor([[100.0, 0.1, 0.1]])

print("原始权重 W:\n", W.numpy())
print("输入激活 X:\n", X.numpy())
print("原始输出 Y = X @ W.T:\n", (X @ W.T).numpy())

# -------- 方法1：Magnitude（只看权重大小）--------
# 计算每个权重的绝对值
magnitude_scores = torch.abs(W)
# 找到最小的那个权重（全局最小）
min_idx = torch.argmin(magnitude_scores)
min_row, min_col = min_idx // W.shape[1], min_idx % W.shape[1]
print(f"\n【Magnitude剪枝决策】剪掉最小的权重: W[{min_row}, {min_col}] = {W[min_row, min_col]}")

mask_magnitude = torch.ones_like(W)
mask_magnitude[min_row, min_col] = 0
W_mag_pruned = W * mask_magnitude
Y_mag = X @ W_mag_pruned.T
print(f"Magnitude剪枝后输出: {Y_mag.numpy()}")
print(f"输出误差 (相比原始): {torch.abs((X @ W.T) - Y_mag).item():.4f}")

# -------- 方法2：Wanda（权重 × 激活范数）--------
# 计算每列的激活范数（这里因为是单条数据，直接用激活值）
# 注意：实际Wanda用的是 该列输入激活的L2范数
activation_norm = torch.norm(X, dim=0)  # 对 batch 维度求 L2 范数
wanda_scores = torch.abs(W) * activation_norm
print(f"\nWanda重要性分数 (|W| * ||X||):\n{wanda_scores.numpy()}")

# 找到Wanda分数最小的权重
min_wanda_idx = torch.argmin(wanda_scores)
min_w_r, min_w_c = min_wanda_idx // W.shape[1], min_wanda_idx % W.shape[1]
print(f"【Wanda剪枝决策】剪掉最小分数的权重: W[{min_w_r}, {min_w_c}] = {W[min_w_r, min_w_c]}")

mask_wanda = torch.ones_like(W)
mask_wanda[min_w_r, min_w_c] = 0
W_wanda_pruned = W * mask_wanda
Y_wanda = X @ W_wanda_pruned.T
print(f"Wanda剪枝后输出: {Y_wanda.numpy()}")
print(f"输出误差 (相比原始): {torch.abs((X @ W.T) - Y_wanda).item():.4f}")

print("\n👉 核心结论：Magnitude错剪了连接小输入(0.1)的0.1，虽然权重小但对输出毫无影响；")
print("           Wanda则准确剪掉了连接大输入(100)的小权重0.2——一旦剪掉，输出直接损失20！")
```



**预期输出解读**： &#x20;

你会发现 Magnitude 剪掉了 `W[0,1]=0.1`（误差极小），而 Wanda 剪掉了 `W[1,0]=0.2`（因为 `0.2 * 100 = 20` 分数高，所以没剪它；实际上剪的是另一个分数更低的权重）。这个实验生动证明了**仅看权重大小会犯致命错误**。







### 三、非结构化 vs 结构化剪枝：矩阵级别的可视化

我们现在切换到一个更大的随机矩阵（模拟一层全连接），直观对比两种剪枝模式产生的稀疏图案。



```python
print("\n" + "="*60)
print("【实验2：非结构化 vs 结构化剪枝（稀疏模式对比）】")

torch.manual_seed(42)
# 模拟一个 8x8 的权重矩阵
W_demo = torch.randn(8, 8)
target_sparsity = 0.5  # 剪掉 50%

# -------- 2.1 非结构化剪枝（全局Top-K）--------
def unstructured_prune(weight, sparsity):
    """非结构化剪枝：全局按绝对值大小剪掉最小的 k%"""
    k = int(weight.numel() * sparsity)
    # 展平后找第k小的阈值
    flat = weight.abs().flatten()
    threshold = torch.kthvalue(flat, k).values
    mask = (weight.abs() > threshold).float()
    return weight * mask, mask

W_unstructured, mask_un = unstructured_prune(W_demo, target_sparsity)
print(f"非结构化剪枝稀疏度: {compute_sparsity(W_unstructured):.2%}")
print("非结构化剪枝掩码(1=保留,0=剪掉):\n", mask_un.numpy().astype(int))

# -------- 2.2 结构化剪枝（按行剪枝）--------
def structured_prune(weight, sparsity):
    """结构化剪枝：按行重要度（L2范数）剪掉整行"""
    row_importance = torch.norm(weight, dim=1)  # 每行的L2范数
    num_rows_to_prune = int(weight.shape[0] * sparsity)
    # 找到重要性最低的几行
    _, indices = torch.topk(row_importance, num_rows_to_prune, largest=False)
    mask = torch.ones_like(weight)
    mask[indices, :] = 0
    return weight * mask, mask

W_structured, mask_st = structured_prune(W_demo, target_sparsity)
print(f"\n结构化剪枝稀疏度: {compute_sparsity(W_structured):.2%}")
print("结构化剪枝掩码(1=保留,0=剪掉):\n", mask_st.numpy().astype(int))

print("\n👉 可视化对比：非结构化剪枝产生零散分布（硬件不友好）；")
print("           结构化剪枝产生连续整行零（硬件友好，但精度损失更大）。")
```







### 四、半结构化稀疏（2:4模式）的完整实现

这是第5课的重头戏。我们实现 Nvidia 的 2:4 稀疏核——对权重矩阵**按列分组（或按行分组）** ，每4个连续元素保留绝对值最大的2个。



```python
print("\n" + "="*60)
print("【实验3：2:4 半结构化稀疏（NVIDIA 硬件友好模式）】")

def prune_2to4(weight):
    """
    对权重矩阵实现 2:4 稀疏（按行分组）
    每行每4个连续元素，保留绝对值最大的2个
    """
    original_shape = weight.shape
    # 展平为 2D: (rows, cols)，方便按行处理
    if weight.dim() == 2:
        rows, cols = weight.shape
        flat = weight.clone()
    else:
        # 处理更高维度（如卷积），先reshape
        flat = weight.view(weight.shape[0], -1)
        rows, cols = flat.shape
    
    # 确保列数能被4整除（否则padding）
    pad_cols = (4 - cols % 4) % 4
    if pad_cols > 0:
        flat = torch.nn.functional.pad(flat, (0, pad_cols), "constant", 0)
        _, cols = flat.shape
    
    mask = torch.ones_like(flat)
    # 每4个一组处理
    for i in range(0, cols, 4):
        block = flat[:, i:i+4]
        # 对每一行，取绝对值最大的两个的索引
        abs_block = torch.abs(block)
        # 找到每行最大的两个值的索引
        top2_idx = torch.topk(abs_block, k=2, dim=1).indices  # [rows, 2]
        # 创建这一块的掩码（默认全0，选中的设为1）
        block_mask = torch.zeros_like(block)
        # 用索引赋值（批量操作）
        for r in range(block.shape[0]):
            block_mask[r, top2_idx[r]] = 1
        mask[:, i:i+4] = block_mask
    
    # 去掉padding
    if pad_cols > 0:
        mask = mask[:, :-pad_cols]
        flat = flat[:, :-pad_cols]
    
    # 恢复原始形状
    if weight.dim() == 2:
        return weight * mask, mask
    else:
        return weight.view(original_shape) * mask.view(original_shape), mask.view(original_shape)

# 测试2:4剪枝
W_test = torch.tensor([
    [1.0, -5.0, 3.0, 0.5],
    [0.1, 0.2, 0.3, 100.0]
])
W_2to4, mask_2to4 = prune_2to4(W_test)
print(f"原始权重:\n{W_test.numpy()}")
print(f"2:4 掩码 (每4个保留2个):\n{mask_2to4.numpy().astype(int)}")
print(f"剪枝后权重:\n{W_2to4.numpy()}")
print(f"每行保留元素数: {mask_2to4.sum(dim=1).numpy()} (应为2)")
```







### 五、剪枝后的精度恢复（微调）

剪枝必然带来精度损失。我们用一个**简化的线性回归任务**展示“剪枝-微调”闭环。



```python
print("\n" + "="*60)
print("【实验4：剪枝后微调恢复（Fine-tuning）】")

# 构造一个简单的回归任务：Y = X @ W_true + noise
torch.manual_seed(123)
input_dim, output_dim = 10, 5
W_true = torch.randn(input_dim, output_dim) * 2
b_true = torch.randn(output_dim) * 0.5

# 生成训练数据
num_samples = 2000
X_data = torch.randn(num_samples, input_dim)
Y_data = X_data @ W_true + b_true + torch.randn(num_samples, output_dim) * 0.1

# 定义一个小型线性模型
class LinearModel(nn.Module):
    def __init__(self, in_dim, out_dim):
        super(LinearModel, self).__init__()
        self.linear = nn.Linear(in_dim, out_dim, bias=True)
    
    def forward(self, x):
        return self.linear(x)

# ----- 步骤1：训练原始模型（FP32）-----
model = LinearModel(input_dim, output_dim).to(device)
optimizer = optim.Adam(model.parameters(), lr=0.01)
criterion = nn.MSELoss()

X_train = X_data.to(device)
Y_train = Y_data.to(device)

for epoch in range(300):
    optimizer.zero_grad()
    loss = criterion(model(X_train), Y_train)
    loss.backward()
    optimizer.step()

with torch.no_grad():
    pred = model(X_train)
    mse_original = criterion(pred, Y_train).item()
print(f"原始模型 MSE: {mse_original:.6f}")

# ----- 步骤2：对权重进行 50% 幅值剪枝 -----
print("执行 50% 非结构化剪枝...")
state_dict = model.state_dict()
weight_key = 'linear.weight'
weight = state_dict[weight_key].clone()

# 生成剪枝掩码
k = int(weight.numel() * 0.5)
threshold = torch.kthvalue(weight.abs().flatten(), k).values
mask = (weight.abs() > threshold).float().to(device)

# 应用剪枝（直接修改权重）
with torch.no_grad():
    state_dict[weight_key] = weight * mask
model.load_state_dict(state_dict)

with torch.no_grad():
    loss_pruned = criterion(model(X_train), Y_train).item()
print(f"剪枝后立即评估 MSE: {loss_pruned:.6f} (精度下降了 {loss_pruned - mse_original:.6f})")

# ----- 步骤3：微调恢复（只更新未被剪枝的权重，固定掩码）-----
print("开始微调（固定掩码，只训练保留的权重）...")
optimizer_finetune = optim.Adam(model.parameters(), lr=0.005)

# 我们用一个技巧：将梯度中对应掩码=0的位置置零，使剪掉的权重永远保持0
def mask_gradients(model, mask):
    """在反向传播后，手动将掩码为0的位置的梯度置零"""
    with torch.no_grad():
        model.linear.weight.grad *= mask

for epoch in range(200):
    optimizer_finetune.zero_grad()
    loss = criterion(model(X_train), Y_train)
    loss.backward()
    # 关键：掩码梯度，让剪掉的权重无法更新
    mask_gradients(model, mask)
    optimizer_finetune.step()
    
    # 额外约束：确保剪掉的权重保持为0（防止数值漂移）
    with torch.no_grad():
        model.linear.weight.data *= mask

with torch.no_grad():
    loss_finetuned = criterion(model(X_train), Y_train).item()
print(f"微调后 MSE: {loss_finetuned:.6f}")
print(f"精度恢复量: {loss_pruned - loss_finetuned:.6f}")
print(f"最终恢复比例: {(1 - (loss_finetuned - mse_original)/(loss_pruned - mse_original)):.1%}")
```

| 本教学代码组件                   | LLM大模型剪枝中的对应操作                  |
| ------------------------- | ------------------------------- |

| `structured_prune` (按行)   | **结构化Head剪枝**（移除整个注意力头）         |
| `prune_2to4`              | **NVIDIA ASP（自动稀疏）库**的底层实现      |
| `mask_gradients` + 微调     | **PrunedLoRA / 剪枝后参数高效微调**      |





**真实LLM剪枝的“三大实操差异”**：



1. **校准数据集**：LLM剪枝（如Wanda/SparseGPT）需要**100-200条校准样本**（从训练集中随机抽取），而非随机生成数据。

2. **逐层剪枝（而非全局）**：LLM对每一层（Attention和FFN）分别计算重要性阈值，因为不同层的权重分布差异极大。

3) **Hessian矩阵计算**：SparseGPT需要计算每层的Hessian逆矩阵（用于误差补偿），这是本教学代码未涉及的进阶内容（涉及 `torch.linalg`）。
