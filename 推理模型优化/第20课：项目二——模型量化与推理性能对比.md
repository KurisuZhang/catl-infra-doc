在上一个项目中，我们完成了7B模型的推理服务部署和性能压测。然而，FP16精度的7B模型（约14GB显存）虽然可以在消费级显卡上运行，但当并发请求增多、上下文长度增加时，**KV Cache的显存爆炸往往导致Batch Size无法提升，显卡算力尚未跑满，显存却已耗尽**。



模型量化正是破解这一困境的核心杠杆，这次的项目会介绍：

4) 建立量化方案的选型框架



**核心工程原则**：量化首先解决的是“能不能部署”，其次才是“跑得有多漂亮”。

#### 1.2 GPTQ量化实战

GPTQ（GPT Quantization）由慕尼黑大学提出，通过**逐层重构策略**优化量化误差——对每个参数进行单独量化后，立即调整相邻参数以补偿精度损失。



**核心参数**：



* `bits`：量化位宽（2-8，常用4）

* `group_size`：分组大小（128为常用值）

* `damp_percent`：阻尼因子（防止Hessian矩阵奇异）

**示例代码**（使用AutoGPTQ库）：



```python
from transformers import AutoTokenizer
from auto_gptq import AutoGPTQForCausalLM, BaseQuantizeConfig
import torch

# 1. 加载原始模型
model_name = "Qwen/Qwen2.5-7B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name, trust_remote_code=True)

# 2. 配置量化参数
quantize_config = BaseQuantizeConfig(
    bits=4,                    # 4-bit量化
    group_size=128,            # 分组大小128
    damp_percent=0.01,         # 阻尼因子
    desc_act=False,            # 是否按列激活排序
    sym=True,                  # 对称量化
)

# 3. 加载模型并进行量化
model = AutoGPTQForCausalLM.from_pretrained(
    model_name,
    quantize_config=quantize_config,
    device_map="auto",
    trust_remote_code=True,
)

# 4. 准备校准数据集（使用少量代表性样本）
from datasets import load_dataset
dataset = load_dataset("wikitext", "wikitext-2-raw-v1", split="train")
calibration_texts = [text for text in dataset["text"][:128] if len(text) > 100]

# 5. 执行量化
model.quantize(
    calibration_texts,
    batch_size=1,
    use_triton=False,          # 可启用Triton加速
)

# 6. 保存量化模型
model.save_quantized("./models/Qwen2.5-7B-Instruct-GPTQ-4bit")
tokenizer.save_pretrained("./models/Qwen2.5-7B-Instruct-GPTQ-4bit")

print("✅ GPTQ量化完成！模型已保存")
```



**GPTQ量化耗时**：7B模型约**30-60分钟**。



#### 1.3 AWQ量化实战

AWQ（Activation-aware Weight Quantization）采用**激活感知的动态量化策略**——通过分析激活分布来识别“关键权重”，仅对约1%的重要参数保留高精度。



**核心优势**：



* 不需要反向传播或校准数据集

* 在保持语义理解能力方面表现突出

* 量化耗时更短（7B模型约15-30分钟）

**示例代码**（使用AutoAWQ库）：



```python
from awq import AutoAWQForCausalLM
from transformers import AutoTokenizer

# 1. 加载原始模型
model_name = "Qwen/Qwen2.5-7B-Instruct"
tokenizer = AutoTokenizer.from_pretrained(model_name, trust_remote_code=True)
model = AutoAWQForCausalLM.from_pretrained(model_name, device_map="auto", trust_remote_code=True)

# 2. 配置量化参数
quant_config = {
    "zero_point": True,        # 启用零点量化
    "q_group_size": 64,        # 分组大小64（AWQ推荐值）
    "w_bit": 4,                # 4-bit量化
    "version": "GEMM",         # 后端选择
}

# 3. 执行量化（AWQ自动使用少量校准数据）
model.quantize(tokenizer, quant_config=quant_config)

# 4. 保存量化模型
model.save_quantized("./models/Qwen2.5-7B-Instruct-AWQ-4bit")
tokenizer.save_pretrained("./models/Qwen2.5-7B-Instruct-AWQ-4bit")

print("✅ AWQ量化完成！模型已保存")
```

### 二、对比FP16/INT8/FP8的推理性能

#### 2.1 性能基准测试框架

以下脚本系统对比FP16、GPTQ-4bit、AWQ-4bit、INT8和FP8五种格式的推理性能：



```python
#!/usr/bin/env python3
"""
量化推理性能对比工具
对比：FP16 / GPTQ-4bit / AWQ-4bit / INT8 / FP8
"""

import time
import json
from vllm import LLM, SamplingParams
import torch


class QuantizationBenchmark:
    """量化方案性能基准测试"""
    
    def __init__(self, output_file="quant_benchmark_results.json"):
        self.results = []
        self.output_file = output_file
        
    def benchmark_model(self, model_path: str, name: str, 
                        quantization: str = None,
                        num_prompts: int = 50,
                        max_tokens: int = 128,
                        input_len: int = 512):
        """测试单个量化方案的性能"""
        print(f"\n📊 测试: {name}")
        print(f"   模型: {model_path}")
        
        # 准备测试数据
        prompts = [
            "What is the capital of France? Explain in detail.",
            "Describe the process of photosynthesis.",
            "Write a short story about a robot learning to paint.",
            "Explain quantum computing to a 10-year-old.",
            "What are the main causes of climate change?",
        ] * (num_prompts // 5 + 1)
        prompts = prompts[:num_prompts]
        
        # 配置vLLM
        llm_kwargs = {
            "model": model_path,
            "gpu_memory_utilization": 0.85,
            "max_model_len": 4096,
            "trust_remote_code": True,
        }
        if quantization:
            llm_kwargs["quantization"] = quantization
        
        # 初始化LLM
        llm = LLM(**llm_kwargs)
        sampling_params = SamplingParams(
            temperature=0.0,  # 确定性输出
            max_tokens=max_tokens,
        )
        
        # 预热
        print("   🔥 预热中...")
        _ = llm.generate(prompts[:5], sampling_params)
        
        # 正式测试
        print(f"   🚀 测试 {num_prompts} 个请求...")
        start_time = time.time()
        outputs = llm.generate(prompts, sampling_params)
        elapsed = time.time() - start_time
        
        # 统计指标
        total_tokens = sum(len(o.outputs[0].token_ids) for o in outputs)
        tokens_per_sec = total_tokens / elapsed
        requests_per_sec = num_prompts / elapsed
        
        # 获取显存信息
        allocated = torch.cuda.memory_allocated() / 1024**3
        reserved = torch.cuda.memory_reserved() / 1024**3
        
        result = {
            "name": name,
            "model_path": model_path,
            "quantization": quantization or "none",
            "num_prompts": num_prompts,
            "max_tokens": max_tokens,
            "total_time": elapsed,
            "total_tokens": total_tokens,
            "tokens_per_sec": tokens_per_sec,
            "requests_per_sec": requests_per_sec,
            "vram_allocated_gb": allocated,
            "vram_reserved_gb": reserved,
        }
        self.results.append(result)
        print(f"   ✅ 完成: {tokens_per_sec:.1f} tok/s, {requests_per_sec:.2f} req/s")
        print(f"   💾 显存: {allocated:.2f} GB (已分配) / {reserved:.2f} GB (预留)")
        
        # 清理
        del llm
        torch.cuda.empty_cache()
        
        return result
    
    def save_results(self):
        """保存测试结果"""
        with open(self.output_file, "w") as f:
            json.dump(self.results, f, indent=2)
        print(f"\n📁 结果已保存至: {self.output_file}")
    
    def print_summary(self):
        """打印对比摘要"""
        print("\n" + "=" * 70)
        print("📊 量化方案性能对比摘要")
        print("=" * 70)
        print(f"{'方案':<15} {'吞吐(tok/s)':<15} {'吞吐(req/s)':<15} {'显存(GB)':<12} {'相对FP16':<10}")
        print("-" * 70)
        
        # 找FP16基线
        fp16_result = next((r for r in self.results if r["quantization"] == "none"), None)
        fp16_tps = fp16_result["tokens_per_sec"] if fp16_result else 1
        
        for r in self.results:
            speedup = r["tokens_per_sec"] / fp16_tps if fp16_result else 1
            print(f"{r['name']:<15} {r['tokens_per_sec']:<15.1f} "
                  f"{r['requests_per_sec']:<15.2f} {r['vram_allocated_gb']:<12.2f} "
                  f"{speedup:<10.2f}x")


# ============ 运行测试 ============
if __name__ == "__main__":
    benchmark = QuantizationBenchmark()
    
    # 测试不同量化方案
    # 注意：需要提前准备好各量化版本的模型
    
    # 1. FP16基线
    benchmark.benchmark_model(
        "./models/Qwen2.5-7B-Instruct",
        "FP16 (baseline)",
        quantization=None
    )
    
    # 2. INT8 (使用vLLM内置)
    benchmark.benchmark_model(
        "./models/Qwen2.5-7B-Instruct",
        "INT8",
        quantization="int8"
    )
    
    # 3. GPTQ-4bit
    benchmark.benchmark_model(
        "./models/Qwen2.5-7B-Instruct-GPTQ-4bit",
        "GPTQ-4bit",
        quantization="gptq"
    )
    
    # 4. AWQ-4bit
    benchmark.benchmark_model(
        "./models/Qwen2.5-7B-Instruct-AWQ-4bit",
        "AWQ-4bit",
        quantization="awq"
    )
    
    # 5. FP8 (仅H100/Ada Lovelace支持)
    try:
        benchmark.benchmark_model(
            "./models/Qwen2.5-7B-Instruct",
            "FP8",
            quantization="fp8"
        )
    except Exception as e:
        print(f"⚠️ FP8测试跳过: {e}")
    
    # 输出结果
    benchmark.print_summary()
    benchmark.save_results()
```

### 三、精度评估与损失分析

#### 3.1 精度评估框架

```python
#!/usr/bin/env python3
"""
量化模型精度评估工具
使用MMLU、GSM8K等基准评估量化精度损失
"""

import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from datasets import load_dataset
import evaluate


class QuantizationAccuracyEvaluator:
    """量化模型精度评估器"""
    
    def __init__(self):
        # 加载评估指标
        self.perplexity = evaluate.load("perplexity")
        
    def evaluate_perplexity(self, model, tokenizer, dataset_name="wikitext", split="test"):
        """计算困惑度（Perplexity）"""
        dataset = load_dataset(dataset_name, "wikitext-2-raw-v1", split=split)
        texts = [t for t in dataset["text"] if len(t) > 50][:100]
        
        results = self.perplexity.compute(
            model_id=None,
            predictions=texts,
            model=model,
            tokenizer=tokenizer,
            add_start_token=True,
        )
        return results["mean_perplexity"]
    
    def evaluate_mmlu(self, model, tokenizer, num_samples=100):
        """简化版MMLU评估（实际应使用完整MMLU基准）"""
        # 这里使用MMLU的简化版本进行演示
        # 生产环境应使用完整的MMLU评估套件
        subjects = ["abstract_algebra", "college_physics", "professional_law"]
        correct = 0
        total = 0
        
        for subject in subjects:
            dataset = load_dataset("cais/mmlu", subject, split="test")
            for sample in dataset.select(range(min(10, len(dataset)))):
                # 构建prompt（简化版）
                prompt = f"Question: {sample['question']}\nChoices: {sample['choices']}\nAnswer:"
                inputs = tokenizer(prompt, return_tensors="pt")
                outputs = model.generate(**inputs, max_new_tokens=10)
                pred = tokenizer.decode(outputs[0], skip_special_tokens=True)
                # 简单匹配（实际应使用更精确的评估）
                correct += 1 if sample['answer'] in pred else 0
                total += 1
        
        return correct / total if total > 0 else 0


def compare_quantization_accuracy():
    """对比不同量化方案的精度"""
    
    models = [
        ("./models/Qwen2.5-7B-Instruct", "FP16"),
        ("./models/Qwen2.5-7B-Instruct-GPTQ-4bit", "GPTQ-4bit"),
        ("./models/Qwen2.5-7B-Instruct-AWQ-4bit", "AWQ-4bit"),
    ]
    
    evaluator = QuantizationAccuracyEvaluator()
    results = []
    
    for model_path, name in models:
        print(f"\n📊 评估: {name}")
        try:
            tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
            model = AutoModelForCausalLM.from_pretrained(
                model_path,
                torch_dtype=torch.float16,
                device_map="auto",
                trust_remote_code=True,
            )
            
            # 计算困惑度
            ppl = evaluator.evaluate_perplexity(model, tokenizer)
            results.append({
                "name": name,
                "perplexity": ppl,
            })
            print(f"   Perplexity: {ppl:.2f}")
            
        except Exception as e:
            print(f"   ❌ 评估失败: {e}")
    
    # 打印对比
    print("\n" + "=" * 50)
    print("📊 精度对比")
    print("=" * 50)
    for r in results:
        print(f"{r['name']}: Perplexity = {r['perplexity']:.2f}")


if __name__ == "__main__":
    compare_quantization_accuracy()
```

#### 3.3 量化精度损失的规避策略

1. **使用更大规模的校准数据集**：增加校准样本量可降低量化误差

2. **分组大小调优**：较小的group\_size（如32）可提升精度，但增加计算开销

3) **混合精度量化**：对关键层（如首尾层）保持FP16，其余层量化

4) **量化感知微调（QAFT）** ：量化后用少量数据微调恢复精度

#### 4.2 场景化选型建议

**场景一：精度优先（研究/医疗/金融）**



* 选型：**FP16** 或 **FP8**（H100环境）

* 理由：FP8在H100上几乎无损（MMLU损失<1%）

**场景二：显存受限（消费级显卡/边缘部署）**



* 选型：**AWQ-4bit** 或 **GPTQ-4bit**

* 理由：4-bit量化将7B模型从14.5GB压缩至4.1-4.2GB

* AWQ在对话场景中精度略优

**场景三：高吞吐生产环境**



* 选型：**AWQ-4bit**（吞吐优先）或 **FP8**（H100环境）

* 理由：AWQ吞吐可达341 tok/s，优于GPTQ的84 tok/s

**场景四：代码/数学推理**



* 选型：**GPTQ-4bit**

* 理由：GPTQ在结构化输出任务中保持更高精度

**场景五：H100极致性能**



* 选型：**FP8**

* 理由：FP8在H100上实现2.3倍加速，并发能力提升5倍

#### 4.3 选型决策树


#### 4.4 量化方案组合优化

最新研究显示，**组合优化可获得最佳效果**：



```plain&#x20;text
最佳实践流程：
LoRA微调 → 合并权重 → 选择性QAT（250步）→ AWQ INT4量化
```



**效果**：



* ROUGE-L: 0.201（与FP16微调基线持平）

```python
#!/usr/bin/env python3
"""
gptq_quantize.py - 使用AutoGPTQ对模型进行4-bit量化
"""

import torch
from transformers import AutoTokenizer, TextGenerationPipeline
from auto_gptq import AutoGPTQForCausalLM, BaseQuantizeConfig
from datasets import load_dataset
import os

# ============ 配置 ============
MODEL_NAME = "Qwen/Qwen2.5-7B-Instruct"  # 或 "meta-llama/Llama-2-7b-hf"
QUANT_DIR = "./models/Qwen2.5-7B-Instruct-GPTQ-4bit"
BITS = 4
GROUP_SIZE = 128
DAMP_PERCENT = 0.01
CALIBRATION_SAMPLES = 128
MAX_SEQ_LEN = 2048

def prepare_calibration_data(tokenizer, num_samples=128):
    """准备校准数据集（使用WikiText-2）"""
    print("📚 加载校准数据集...")
    dataset = load_dataset("wikitext", "wikitext-2-raw-v1", split="train")
    
    # 过滤短文本并截断
    texts = []
    for text in dataset["text"]:
        if len(text) > 100:
            texts.append(text[:MAX_SEQ_LEN])
        if len(texts) >= num_samples:
            break
    
    # Tokenize
    encodings = tokenizer(
        texts,
        truncation=True,
        padding=False,
        max_length=MAX_SEQ_LEN,
        return_tensors="pt"
    )
    
    # 构造校准数据格式（AutoGPTQ要求）
    examples = []
    for i in range(len(encodings["input_ids"])):
        examples.append({
            "input_ids": encodings["input_ids"][i],
            "attention_mask": encodings["attention_mask"][i]
        })
    
    print(f"✅ 校准数据准备完成: {len(examples)} 条样本")
    return examples

def quantize_gptq():
    """执行GPTQ量化"""
    print("=" * 60)
    print("🚀 GPTQ 4-bit 量化")
    print("=" * 60)
    print(f"模型: {MODEL_NAME}")
    print(f"量化位宽: {BITS}bit, 分组大小: {GROUP_SIZE}")
    
    # 1. 加载Tokenizer
    tokenizer = AutoTokenizer.from_pretrained(
        MODEL_NAME,
        trust_remote_code=True
    )
    
    # 2. 配置量化参数
    quantize_config = BaseQuantizeConfig(
        bits=BITS,
        group_size=GROUP_SIZE,
        damp_percent=DAMP_PERCENT,
        desc_act=False,          # 是否按列激活排序
        sym=True,                # 对称量化
        model_file_base_name="model"
    )
    
    print(f"量化配置: {quantize_config.__dict__}")
    
    # 3. 加载模型
    print("⏳ 加载原始模型...")
    model = AutoGPTQForCausalLM.from_pretrained(
        MODEL_NAME,
        quantize_config=quantize_config,
        device_map="auto",
        trust_remote_code=True,
        torch_dtype=torch.float16,
    )
    
    # 4. 准备校准数据
    calibration_examples = prepare_calibration_data(tokenizer, CALIBRATION_SAMPLES)
    
    # 5. 执行量化
    print("⏳ 执行GPTQ量化（预计30-60分钟）...")
    model.quantize(
        calibration_examples,
        batch_size=1,
        use_triton=False,        # 可启用Triton加速
        cache_examples_on_gpu=False,
    )
    
    # 6. 保存量化模型
    print(f"💾 保存量化模型到: {QUANT_DIR}")
    os.makedirs(QUANT_DIR, exist_ok=True)
    model.save_quantized(QUANT_DIR)
    tokenizer.save_pretrained(QUANT_DIR)
    
    print("✅ GPTQ量化完成！")
    print(f"   模型保存在: {QUANT_DIR}")
    
    # 7. 验证加载
    print("🔍 验证量化模型加载...")
    test_model = AutoGPTQForCausalLM.from_quantized(
        QUANT_DIR,
        device_map="auto",
        use_triton=False,
    )
    print("✅ 量化模型加载成功")

if __name__ == "__main__":
    quantize_gptq()
```

```python
#!/usr/bin/env python3
"""
quant_benchmark.py - 量化模型推理性能对比
对比：FP16 / GPTQ-4bit / AWQ-4bit / INT8 / FP8
"""

import time
import json
import torch
from vllm import LLM, SamplingParams
from typing import List, Dict
import statistics


class QuantizationBenchmark:
    """量化方案性能基准测试"""
    
    def __init__(self, output_file="quant_benchmark_results.json"):
        self.results = []
        self.output_file = output_file
    
    def benchmark_model(self, model_path: str, name: str, 
                        quantization: str = None,
                        num_prompts: int = 50,
                        max_tokens: int = 128) -> Dict:
        """测试单个量化方案的性能"""
        print(f"\n📊 测试: {name}")
        print(f"   模型: {model_path}")
        
        # 准备测试数据
        prompts = [
            "What is the capital of France? Explain in detail.",
            "Describe the process of photosynthesis in simple terms.",
            "Write a short story about a robot learning to paint.",
            "Explain quantum computing to a 10-year-old.",
            "What are the main causes of climate change?",
            "How does a transformer model work?",
            "What is the difference between AI and ML?",
            "Write a haiku about the ocean.",
        ] * (num_prompts // 8 + 1)
        prompts = prompts[:num_prompts]
        
        # 配置vLLM
        llm_kwargs = {
            "model": model_path,
            "gpu_memory_utilization": 0.85,
            "max_model_len": 4096,
            "trust_remote_code": True,
        }
        if quantization:
            llm_kwargs["quantization"] = quantization
        
        # 初始化LLM
        try:
            llm = LLM(**llm_kwargs)
        except Exception as e:
            print(f"   ❌ 模型加载失败: {e}")
            return {"name": name, "error": str(e)}
        
        sampling_params = SamplingParams(
            temperature=0.0,       # 确定性输出
            max_tokens=max_tokens,
        )
        
        # 预热
        print("   🔥 预热中...")
        _ = llm.generate(prompts[:5], sampling_params)
        
        # 正式测试
        print(f"   🚀 测试 {num_prompts} 个请求...")
        start_time = time.time()
        outputs = llm.generate(prompts, sampling_params)
        elapsed = time.time() - start_time
        
        # 统计指标
        total_tokens = sum(len(o.outputs[0].token_ids) for o in outputs)
        tokens_per_sec = total_tokens / elapsed
        requests_per_sec = num_prompts / elapsed
        
        # 计算TTFT和TPOT（近似）
        ttft_list = []
        tpot_list = []
        for o in outputs:
            # 模拟TTFT（实际应从引擎获取）
            ttft_list.append(0.05)  # 占位
            tpot_list.append(0.02)
        
        # 获取显存信息
        if torch.cuda.is_available():
            allocated = torch.cuda.memory_allocated() / 1024**3
            reserved = torch.cuda.memory_reserved() / 1024**3
        else:
            allocated = 0
            reserved = 0
        
        result = {
            "name": name,
            "model_path": model_path,
            "quantization": quantization or "none",
            "num_prompts": num_prompts,
            "max_tokens": max_tokens,
            "total_time": elapsed,
            "total_tokens": total_tokens,
            "tokens_per_sec": tokens_per_sec,
            "requests_per_sec": requests_per_sec,
            "vram_allocated_gb": allocated,
            "vram_reserved_gb": reserved,
            "avg_ttft_ms": statistics.mean(ttft_list) * 1000 if ttft_list else 0,
            "avg_tpot_ms": statistics.mean(tpot_list) * 1000 if tpot_list else 0,
        }
        self.results.append(result)
        print(f"   ✅ 完成: {tokens_per_sec:.1f} tok/s, {requests_per_sec:.2f} req/s")
        print(f"   💾 显存: {allocated:.2f} GB")
        
        # 清理
        del llm
        if torch.cuda.is_available():
            torch.cuda.empty_cache()
        
        return result
    
    def run_all_benchmarks(self):
        """运行所有量化方案的基准测试"""
        print("=" * 70)
        print("📊 量化推理性能基准测试")
        print("=" * 70)
        
        # 1. FP16基线
        self.benchmark_model(
            "./models/Qwen2.5-7B-Instruct",
            "FP16 (baseline)",
            quantization=None
        )
        
        # 2. INT8（使用vLLM内置量化）
        try:
            self.benchmark_model(
                "./models/Qwen2.5-7B-Instruct",
                "INT8",
                quantization="int8"
            )
        except Exception as e:
            print(f"⚠️ INT8测试跳过: {e}")
        
        # 3. GPTQ-4bit
        try:
            self.benchmark_model(
                "./models/Qwen2.5-7B-Instruct-GPTQ-4bit",
                "GPTQ-4bit",
                quantization="gptq"
            )
        except Exception as e:
            print(f"⚠️ GPTQ测试跳过: {e}")
        
        # 4. AWQ-4bit
        try:
            self.benchmark_model(
                "./models/Qwen2.5-7B-Instruct-AWQ-4bit",
                "AWQ-4bit",
                quantization="awq"
            )
        except Exception as e:
            print(f"⚠️ AWQ测试跳过: {e}")
        
        # 5. FP8 (仅H100/Ada Lovelace支持)
        try:
            self.benchmark_model(
                "./models/Qwen2.5-7B-Instruct",
                "FP8",
                quantization="fp8"
            )
        except Exception as e:
            print(f"⚠️ FP8测试跳过: {e}")
    
    def print_summary(self):
        """打印对比摘要"""
        print("\n" + "=" * 80)
        print("📊 量化方案性能对比摘要")
        print("=" * 80)
        print(f"{'方案':<16} {'吞吐(tok/s)':<14} {'吞吐(req/s)':<14} "
              f"{'显存(GB)':<12} {'相对FP16':<10}")
        print("-" * 80)
        
        # 找FP16基线
        fp16_result = next((r for r in self.results if r.get("quantization") == "none"), None)
        fp16_tps = fp16_result.get("tokens_per_sec", 1) if fp16_result else 1
        
        for r in self.results:
            if "error" in r:
                print(f"{r['name']:<16} ❌ 加载失败")
                continue
            speedup = r["tokens_per_sec"] / fp16_tps if fp16_tps > 0 else 1
            print(f"{r['name']:<16} {r['tokens_per_sec']:<14.1f} "
                  f"{r['requests_per_sec']:<14.2f} "
                  f"{r['vram_allocated_gb']:<12.2f} "
                  f"{speedup:<10.2f}x")
    
    def save_results(self):
        """保存测试结果"""
        with open(self.output_file, "w") as f:
            json.dump(self.results, f, indent=2)
        print(f"\n📁 结果已保存至: {self.output_file}")


if __name__ == "__main__":
    benchmark = QuantizationBenchmark()
    benchmark.run_all_benchmarks()
    benchmark.print_summary()
    benchmark.save_results()
```

```python
#!/usr/bin/env python3
"""
quant_accuracy.py - 量化模型精度评估
使用困惑度(Perplexity)评估量化精度损失
"""

import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from datasets import load_dataset
import evaluate
import math


class QuantizationAccuracyEvaluator:
    """量化模型精度评估器"""
    
    def __init__(self):
        self.perplexity_metric = evaluate.load("perplexity")
    
    def evaluate_perplexity(self, model_path: str, name: str, 
                            num_samples: int = 100) -> float:
        """计算困惑度（Perplexity）"""
        print(f"\n📊 评估: {name}")
        
        try:
            tokenizer = AutoTokenizer.from_pretrained(
                model_path,
                trust_remote_code=True
            )
            model = AutoModelForCausalLM.from_pretrained(
                model_path,
                torch_dtype=torch.float16,
                device_map="auto",
                trust_remote_code=True,
            )
        except Exception as e:
            print(f"   ❌ 模型加载失败: {e}")
            return float('inf')
        
        # 加载测试数据
        dataset = load_dataset("wikitext", "wikitext-2-raw-v1", split="test")
        texts = [t for t in dataset["text"] if len(t) > 50][:num_samples]
        
        # 计算困惑度
        results = self.perplexity_metric.compute(
            predictions=texts,
            model=model,
            tokenizer=tokenizer,
            add_start_token=True,
            batch_size=4,
        )
        
        ppl = results.get("mean_perplexity", float('inf'))
        print(f"   Perplexity: {ppl:.2f}")
        
        # 清理
        del model
        if torch.cuda.is_available():
            torch.cuda.empty_cache()
        
        return ppl
    
    def run_accuracy_comparison(self):
        """运行精度对比"""
        print("=" * 70)
        print("📊 量化模型精度对比（困惑度）")
        print("=" * 70)
        
        models = [
            ("./models/Qwen2.5-7B-Instruct", "FP16 (baseline)"),
            ("./models/Qwen2.5-7B-Instruct-GPTQ-4bit", "GPTQ-4bit"),
            ("./models/Qwen2.5-7B-Instruct-AWQ-4bit", "AWQ-4bit"),
        ]
        
        results = {}
        for model_path, name in models:
            ppl = self.evaluate_perplexity(model_path, name)
            results[name] = ppl
        
        # 打印对比
        print("\n" + "=" * 70)
        print("📊 精度对比结果")
        print("=" * 70)
        
        baseline = results.get("FP16 (baseline)", float('inf'))
        for name, ppl in results.items():
            if ppl == float('inf'):
                print(f"   {name}: ❌ 评估失败")
                continue
            if name == "FP16 (baseline)":
                print(f"   {name}: {ppl:.2f} (基线)")
            else:
                delta = ppl - baseline
                print(f"   {name}: {ppl:.2f} (Δ = {delta:+.2f}, "
                      f"损失 {delta/baseline*100:.1f}%)")


if __name__ == "__main__":
    evaluator = QuantizationAccuracyEvaluator()
    evaluator.run_accuracy_comparison()
```

```plain&#x20;text
│   ├── 追求极致吞吐 + 精度无损 → FP8
│   └── 追求极致压缩 → AWQ-4bit
│
├── 消费级GPU (RTX 30/40系列)
│   ├── 对话/文本生成 → AWQ-4bit（速度+精度平衡）
│   ├── 代码/数学推理 → GPTQ-4bit（精度更稳）
│   └── 显存极度受限（<8GB）→ 4-bit量化
│
└── CPU/边缘设备
    └── GGUF（llama.cpp生态）
```



#### 6.3 量化精度损失的规避策略

1. **使用更大规模的校准数据集**：增加校准样本量可降低量化误差

2. **分组大小调优**：较小的group\_size（如32）可提升精度，但增加计算开销

3) **混合精度量化**：对关键层（首尾层）保持FP16，其余层量化

4) **量化感知微调（QAFT）** ：量化后用少量数据微调恢复精度
