### 二、结构化日志配置（推理服务端）

在vLLM中，可以通过自定义 `RequestLogger` 输出结构化JSON日志，便于后续聚合和分析。



#### 2.1 vLLM结构化日志配置

```python
#!/usr/bin/env python3
"""
推理服务结构化日志配置
为vLLM推理服务添加JSON格式的结构化日志，包含trace_id、model_id等关键字段
"""

import json
import logging
import time
from contextvars import ContextVar
from typing import Optional

# 创建上下文变量存储trace_id（支持跨协程传递）
trace_id_ctx: ContextVar[str] = ContextVar("trace_id", default="")

class StructuredJSONFormatter(logging.Formatter):
    """将日志格式化为JSON结构，便于日志聚合系统解析"""
    
    def format(self, record: logging.LogRecord) -> str:
        log_entry = {
            "timestamp": time.time(),
            "time_str": time.strftime("%Y-%m-%dT%H:%M:%S.%fZ", time.gmtime()),
            "level": record.levelname,
            "module": record.module,
            "funcName": record.funcName,
            "lineno": record.lineno,
            "trace_id": trace_id_ctx.get(),  # 从上下文提取trace_id
            "message": record.getMessage(),
        }
        # 如果有额外的属性，合并到日志中
        if hasattr(record, "extra_fields"):
            log_entry.update(record.extra_fields)
        return json.dumps(log_entry)

def setup_structured_logging():
    """配置结构化日志"""
    logger = logging.getLogger("vllm_inference")
    logger.setLevel(logging.INFO)
    
    handler = logging.StreamHandler()
    handler.setFormatter(StructuredJSONFormatter())
    logger.addHandler(handler)
    
    return logger

def log_inference_request(logger, request_id: str, prompt_len: int, model_id: str):
    """记录推理请求日志（包含关键指标）"""
    trace_id_ctx.set(request_id)
    extra = {
        "request_id": request_id,
        "prompt_len": prompt_len,
        "model_id": model_id,
    }
    # 设置extra_fields供formatter使用
    logger.info(f"Inference request received", extra={"extra_fields": extra})

def log_inference_response(logger, request_id: str, output_len: int, 
                           ttft_ms: float, tpot_ms: float, cache_hit: bool,
                           status: str = "success"):
    """记录推理响应日志"""
    extra = {
        "request_id": request_id,
        "output_len": output_len,
        "ttft_ms": ttft_ms,
        "tpot_ms": tpot_ms,
        "cache_hit": cache_hit,
        "status": status,
    }
    logger.info(f"Inference response completed", extra={"extra_fields": extra})

# ============ 使用示例 ============
if __name__ == "__main__":
    logger = setup_structured_logging()
    
    # 模拟推理请求
    req_id = f"req_{int(time.time())}_{12345}"
    log_inference_request(logger, req_id, 2048, "llama-2-7b")
    
    # 模拟推理完成
    time.sleep(0.1)  # 模拟计算
    log_inference_response(logger, req_id, 256, 150.5, 45.2, True)
    
    # 输出示例:
    # {"timestamp": 1734567890.123, "level": "INFO", "trace_id": "req_1734567890_12345", 
    #  "message": "Inference request received", "request_id": "...", "prompt_len": 2048, ...}
```

#### 4.2 ClickHouse日志分析

对于需要复杂聚合分析的高性能查询场景，可引入ClickHouse作为日志分析引擎。



```sql
-- 创建推理日志表
CREATE TABLE llm_inference_logs (
    trace_id String,
    session_id String,
    model_id String,
    prompt_len UInt16,
    output_len UInt16,
    ttft_ms Float32,
    tpot_ms Float32,
    cache_hit UInt8,
    status String,
    timestamp DateTime
) ENGINE = MergeTree()
ORDER BY (timestamp, model_id, status);

-- 查询分析：按时间统计平均TTFT
SELECT 
    toStartOfMinute(timestamp) AS minute,
    model_id,
    avg(ttft_ms) AS avg_ttft,
    quantile(0.99)(ttft_ms) AS p99_ttft,
    count() AS request_count
FROM llm_inference_logs
WHERE timestamp > now() - INTERVAL 1 HOUR
GROUP BY minute, model_id
ORDER BY minute DESC;
```







### 五、异常检测引擎

#### 5.1 基于Isolation Forest的日志异常检测

使用Isolation Forest对日志特征进行异常检测：



```python
import pandas as pd
import numpy as np
from sklearn.ensemble import IsolationForest
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.cluster import KMeans
import json
from typing import List, Dict

class LogAnomalyDetector:
    """基于机器学习的日志异常检测器"""
    
    def __init__(self, contamination: float = 0.05):
        self.contamination = contamination
        self.model = IsolationForest(
            contamination=contamination,
            random_state=42
        )
        self.feature_columns = ['ttft_ms', 'tpot_ms', 'prompt_len', 'output_len']
    
    def extract_features(self, logs: List[Dict]) -> pd.DataFrame:
        """从结构化日志中提取数值特征"""
        df = pd.DataFrame(logs)
        # 只保留数值列
        numeric_cols = [c for c in self.feature_columns if c in df.columns]
        return df[numeric_cols].fillna(0)
    
    def fit(self, logs: List[Dict]):
        """训练异常检测模型"""
        X = self.extract_features(logs)
        self.model.fit(X)
        return self
    
    def predict(self, logs: List[Dict]) -> List[bool]:
        """预测日志是否为异常（True=异常）"""
        X = self.extract_features(logs)
        predictions = self.model.predict(X)
        # Isolation Forest: -1表示异常，1表示正常
        return [p == -1 for p in predictions]

# 使用示例
logs = [
    {"trace_id": "1", "ttft_ms": 150, "tpot_ms": 45, "prompt_len": 2048, "output_len": 256},
    {"trace_id": "2", "ttft_ms": 120, "tpot_ms": 40, "prompt_len": 1024, "output_len": 128},
    {"trace_id": "3", "ttft_ms": 3000, "tpot_ms": 200, "prompt_len": 8192, "output_len": 1024},  # 异常
    {"trace_id": "4", "ttft_ms": 180, "tpot_ms": 50, "prompt_len": 2048, "output_len": 300},
]

detector = LogAnomalyDetector(contamination=0.1)
detector.fit(logs)
results = detector.predict(logs)
print(f"异常检测结果: {results}")  # [False, False, True, False]
```



#### 5.2 日志聚类（模式识别）

使用K-Means将相似日志自动归类，识别异常模式：



```python
def cluster_logs(log_messages: List[str], n_clusters: int = 3):
    """对日志消息进行聚类，识别不同模式"""
    vectorizer = TfidfVectorizer(stop_words="english", max_features=100)
    X = vectorizer.fit_transform(log_messages)
    
    kmeans = KMeans(n_clusters=n_clusters, random_state=42)
    labels = kmeans.fit_predict(X)
    
    # 打印每个聚类的中心词
    feature_names = vectorizer.get_feature_names_out()
    for i in range(n_clusters):
        center = kmeans.cluster_centers_[i]
        top_idx = center.argsort()[-5:][::-1]
        top_words = [feature_names[idx] for idx in top_idx]
        print(f"Cluster {i}: {' '.join(top_words)}")
    
    return labels

# 推理日志聚类示例
log_msgs = [
    "Request 1 completed with TTFT=150ms, cache hit",
    "Request 2 completed with TTFT=120ms, cache hit",
    "Request 3 failed: CUDA out of memory",
    "Request 4 completed with TTFT=180ms, cache miss",
    "Request 5 failed: Connection timeout",
    "Request 6 completed with TTFT=160ms, cache hit",
    "Request 7 failed: CUDA out of memory",
]
cluster_logs(log_msgs, n_clusters=3)
```



#### 5.3 规则驱动的异常检测

基于SLO阈值的规则引擎是最实用的异常检测方式：



```python
import time
from collections import defaultdict
from typing import Dict, List, Tuple

class RuleBasedAnomalyDetector:
    """基于规则的异常检测器"""
    
    def __init__(self):
        self.rules = {}
        self.error_counter = defaultdict(int)
        self.window_start = time.time()
        self.window_size = 60  # 60秒滑动窗口
    
    def add_rule(self, name: str, condition: callable, severity: str = "warning"):
        """添加检测规则"""
        self.rules[name] = {"condition": condition, "severity": severity}
    
    def check(self, log: Dict) -> List[Tuple[str, str, str]]:
        """
        检查单条日志是否触发规则
        返回: [(rule_name, severity, description), ...]
        """
        alerts = []
        now = time.time()
        
        for rule_name, rule in self.rules.items():
            try:
                if rule["condition"](log):
                    alerts.append((rule_name, rule["severity"], log.get("message", "")))
            except Exception:
                pass
        
        # 滑动窗口错误计数
        if log.get("status") == "error" or log.get("level") == "ERROR":
            self.error_counter[log.get("trace_id", "unknown")] += 1
        
        return alerts
    
    def get_window_stats(self) -> Dict:
        """获取滑动窗口统计"""
        # 清理过期数据
        return {
            "error_count": sum(self.error_counter.values()),
            "unique_errors": len(self.error_counter)
        }

# 配置规则
detector = RuleBasedAnomalyDetector()

# TTFT超时规则
detector.add_rule(
    "TTFT_SLO_Violation",
    lambda log: log.get("ttft_ms", 0) > 800,
    "critical"
)

# TPOT超时规则
detector.add_rule(
    "TPOT_SLO_Violation", 
    lambda log: log.get("tpot_ms", 0) > 150,
    "critical"
)

# 缓存命中率低（异常模式）
detector.add_rule(
    "LowCacheHit",
    lambda log: log.get("cache_hit") is False and log.get("cache_hit_rate", 1) < 0.3,
    "warning"
)

# 使用示例
test_log = {"trace_id": "req_123", "ttft_ms": 1200, "tpot_ms": 60, "status": "success"}
alerts = detector.check(test_log)
print(f"触发告警: {alerts}")  # [('TTFT_SLO_Violation', 'critical', '')]
```
